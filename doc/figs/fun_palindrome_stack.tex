    \newcommand{\xStep}[0]{3cm}

    \newcommand{\elementheight}[0]{ 6mm }
    \newcommand{\elementwidth}[0]{ 14mm }
    \newcommand{\newframe}[5]{
      \coordinate (tmp) at (#2);
      \node[very thick,draw=purple,anchor=south,minimum width=\elementwidth,minimum height=3*\elementheight]
        (#1) at (tmp) {};
      \node[draw=purple,anchor=south,minimum width=\elementwidth,minimum height=  \elementheight]
        (#1 input) at ([yshift=2*\elementheight]tmp) {\texttt{#5}};
      \node[draw=purple,anchor=south,minimum width=\elementwidth,minimum height=  \elementheight]
        (#1 left) at ([yshift=1*\elementheight]tmp) {\texttt{#4}};
      \node[draw=purple,anchor=south,minimum width=\elementwidth,minimum height=  \elementheight]
        (#1 right) at ([yshift=0*\elementheight]tmp) {\texttt{#3}};
    }
    \newcommand{\newSmallframe}[3]{
      \coordinate (tmp) at (#2);
      \node[very thick,draw=purple,anchor=south,minimum width=\elementwidth,minimum height=1*\elementheight]
        (#1) at (tmp) {};
      \node[draw=purple,anchor=south,minimum width=\elementwidth,minimum height=  \elementheight]
        (#1 input) at ([yshift=0*\elementheight]tmp) {\texttt{#3}};
    }
    \newcommand{\newBaseline}[1]{
      \coordinate (tmp) at (#1);
      \draw[ultra thick,draw=purple]
           ([xshift=-0.8*\elementwidth,yshift=0*\elementheight]tmp)
        -- ([xshift= 0.8*\elementwidth,yshift=0*\elementheight]tmp)
      ;
    }
    \newcommand{\newTransition}[2]{
      \coordinate (src) at (#1);
      \coordinate (dst) at (#2);
      \draw[dedge]
           ([xshift= 0.8*\elementwidth,yshift=1.5*\elementheight]src)
        -- ([xshift=-0.8*\elementwidth,yshift=1.5*\elementheight]dst)
      ;
    }
